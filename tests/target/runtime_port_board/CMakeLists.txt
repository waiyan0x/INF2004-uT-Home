cmake_minimum_required(VERSION 3.18)
# qualify_kernel.py stages this project beside the pinned kernel source.
set(EXTRA_COMPONENT_DIRS ../source/components)
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;parser.defaults;audit.defaults")
if(UTHOME_BLE_GATT_PROBE AND NOT UTHOME_BLE_PROBE)
    message(FATAL_ERROR "GATT diagnostic requires the isolated BLE probe")
endif()
if(UTHOME_BLE_STATIC_HOST AND NOT UTHOME_BLE_PROBE)
    message(FATAL_ERROR "Static host experiment requires the isolated BLE probe")
endif()
if(UTHOME_MODEL_PROBE)
    if(UTHOME_WIFI_PROBE OR UTHOME_BLE_PROBE OR UTHOME_CONFIGURATION_APP OR UTHOME_NATIVE_UART_FIXTURE OR NOT UTHOME_STATIC_NET_QUEUES)
        message(FATAL_ERROR "Model probe requires isolated runtime and fixed queues")
    endif()
endif()
if(UTHOME_CONFIGURATION_APP)
    list(APPEND SDKCONFIG_DEFAULTS "configuration.defaults")
endif()
if(UTHOME_WIFI_PROBE)
    if(UTHOME_BLE_PROBE OR UTHOME_CONFIGURATION_APP OR UTHOME_NATIVE_UART_FIXTURE OR NOT UTHOME_STATIC_NET_QUEUES)
        message(FATAL_ERROR "Wi-Fi probe requires isolated runtime and fixed queues")
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "wifi.defaults")
endif()
if(UTHOME_BLE_PROBE)
    if(UTHOME_WIFI_PROBE OR UTHOME_CONFIGURATION_APP OR UTHOME_NATIVE_UART_FIXTURE OR NOT UTHOME_STATIC_NET_QUEUES)
        message(FATAL_ERROR "BLE probe requires isolated runtime and fixed queues")
    endif()
    list(APPEND SDKCONFIG_DEFAULTS "ble.defaults")
    if(UTHOME_BLE_GATT_PROBE)
        list(APPEND SDKCONFIG_DEFAULTS "ble-gatt.defaults")
    endif()
    if(UTHOME_BLE_STATIC_HOST)
        list(APPEND SDKCONFIG_DEFAULTS "ble-static.defaults")
    endif()
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(uthome_runtime_port)
